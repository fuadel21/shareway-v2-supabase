import { serve } from 'https://deno.land/std@0.168.0/http/server.ts'
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'
import { getCorsHeaders } from '../_shared/cors.ts'

const jsonResponse = (payload: unknown, headers: Record<string, string>, status = 200) => new Response(
  JSON.stringify(payload),
  { status, headers: { ...headers, 'Content-Type': 'application/json' } },
)

// This file is kept as a non-deployed compatibility source. If the function is
// enabled, it delegates to the same admin-gated SQL read model as the dashboard
// instead of projecting commissions from rides.price.
serve(async (req) => {
  const corsHeaders = getCorsHeaders(req)
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })
  if (req.method !== 'POST') return jsonResponse({ error: 'Método no permitido' }, corsHeaders, 405)

  const authorization = req.headers.get('Authorization')
  if (!authorization) return jsonResponse({ error: 'No autorizado' }, corsHeaders, 401)

  try {
    const supabase = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_ANON_KEY') ?? '',
      { global: { headers: { Authorization: authorization } } },
    )
    const { data, error } = await supabase.rpc('get_platform_admin_stats')
    if (error) return jsonResponse({ error: error.message }, corsHeaders, 403)

    return jsonResponse(data, corsHeaders)
  } catch (error) {
    return jsonResponse({
      error: error instanceof Error ? error.message : 'No se pudieron cargar las estadísticas.',
    }, corsHeaders, 500)
  }
})
