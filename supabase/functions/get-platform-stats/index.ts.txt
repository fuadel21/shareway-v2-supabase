import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';
import { corsHeaders } from '../_shared/cors.ts';

// Define a constant for the platform's commission rate (e.g., 15%)
// IMPORTANT: Adjust this value to match your actual commission rate.
const COMMISSION_RATE = 0.15;

Deno.serve(async (req) => {
  // This is needed to handle the preflight request for CORS.
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    // Create a Supabase client with the service_role key to bypass RLS.
    const supabaseAdmin = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    );

    // We run the queries in parallel to be more efficient.
    const [
      { count: totalUsers, error: usersError },
      { data: ridesData, error: ridesError },
      { data: recentRides, error: recentError }
    ] = await Promise.all([
      // Query 1: Get the total number of users.
      supabaseAdmin.from('profiles').select('*', { count: 'exact', head: true }),
      // Query 2: Get the price of all completed rides.
      supabaseAdmin.from('rides').select('price').eq('status', 'completed'),
      // Query 3: Get recent completed rides for transactions list
      supabaseAdmin.from('rides')
        .select('id, price, completed_at, passengers')
        .eq('status', 'completed')
        .order('completed_at', { ascending: false })
        .limit(10)
    ]);

    if (usersError) throw usersError;
    if (ridesError) throw ridesError;
    if (recentError) throw recentError;

    // --- Calculate statistics ---
    const totalCompletedRides = ridesData?.length || 0;
    const grossTotalVolume = ridesData?.reduce((sum, ride) => sum + (ride.price || 0), 0) || 0;
    
    // In this simulation, we consider all completed volume * rate as uncollected till payout
    const totalUncollectedFees = grossTotalVolume * COMMISSION_RATE;

    const recentFeeTransactions = (recentRides || []).map(ride => ({
        id: `fee_${ride.id}`,
        date: ride.completed_at,
        description: `Comisión viaje ${ride.id.substring(0,8)}`,
        rideId: ride.id,
        amount: (ride.price || 0) * COMMISSION_RATE
    }));

    const stats = {
      totalUsers,
      totalCompletedRides,
      grossTotalVolume,
      totalUncollectedFees,
      recentFeeTransactions
    };

    // Return the statistics in a JSON response.
    return new Response(JSON.stringify({ success: true, stats }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      status: 200,
    });

  } catch (error) {
    // If any error occurs, return an error response.
    return new Response(JSON.stringify({ success: false, message: error.message }), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      status: 500,
    });
  }
});
